if(CONFIG_ZMK_HIRES_DIAL)
  get_property(equinox_libs GLOBAL PROPERTY ZEPHYR_LIBS)
  set(pat_found FALSE)
  foreach(lib IN LISTS equinox_libs)
    get_target_property(pat_sources ${lib} SOURCES)
    get_target_property(pat_source_dir ${lib} SOURCE_DIR)
    foreach(source IN LISTS pat_sources)
      get_filename_component(pat_absolute "${source}" ABSOLUTE BASE_DIR "${pat_source_dir}")
      if(pat_absolute MATCHES "zmk-driver-hires-dial/src/hires_dial.c$")
        set(pat_found TRUE)
        set(patched_pat "${CMAKE_CURRENT_BINARY_DIR}/equinox_hires_dial.c")
        set(pat_script "${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch_pat_address.py")
        execute_process(COMMAND ${PYTHON_EXECUTABLE} ${pat_script} ${pat_absolute} ${patched_pat}
                        RESULT_VARIABLE pat_result)
        if(NOT pat_result EQUAL 0)
          message(FATAL_ERROR "PAT address patch requires source review")
        endif()
        list(REMOVE_ITEM pat_sources ${source})
        list(APPEND pat_sources ${patched_pat})
        set_property(TARGET ${lib} PROPERTY SOURCES ${pat_sources})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${pat_absolute} ${pat_script})
      endif()
    endforeach()
  endforeach()
  if(NOT pat_found)
    message(FATAL_ERROR "PAT driver not found for address patch")
  endif()
  # The pinned dial module uses ZMK behavior headers without exporting this path.
  zephyr_include_directories(${APPLICATION_SOURCE_DIR}/include)
  # Adapt the upstream dial API name to the pinned cormoran ZMK endpoint API.
  if(NOT CONFIG_ZMK_NEMU_LINK)
    zephyr_compile_definitions(zmk_endpoints_selected=zmk_endpoint_get_selected)
  endif()
endif()

# Scope the bootloader handoff fix to the nRF52840 nemu keyboard images.
# Driver targets are created after external modules, so substitute at the end
# of configuration. The patch script rejects unreviewed upstream source.
if(CONFIG_ZMK_NEMU_LINK AND CONFIG_SOC_NRF52840)
  set_property(GLOBAL PROPERTY EQUINOX_BOOT_CLOCK_PATCH_SCRIPT
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch_nemu_boot_clock.py")
  function(equinox_patch_boot_clock)
    get_property(patch_script GLOBAL PROPERTY EQUINOX_BOOT_CLOCK_PATCH_SCRIPT)
    set(original "${ZEPHYR_BASE}/drivers/clock_control/clock_control_nrf.c")
    set(patched "${PROJECT_BINARY_DIR}/equinox_nemu_clock_control.c")
    execute_process(COMMAND ${PYTHON_EXECUTABLE} ${patch_script} ${original} ${patched}
      RESULT_VARIABLE patch_result)
    if(NOT patch_result EQUAL 0)
      message(FATAL_ERROR "Equinox boot clock patch requires source review")
    endif()
    get_target_property(clock_sources drivers__clock_control SOURCES)
    set(new_sources)
    set(replaced_count 0)
    foreach(source IN LISTS clock_sources)
      if(source MATCHES "clock_control_nrf.c$")
        list(APPEND new_sources "${patched}")
        math(EXPR replaced_count "${replaced_count} + 1")
      else()
        list(APPEND new_sources "${source}")
      endif()
    endforeach()
    if(NOT replaced_count EQUAL 1)
      message(FATAL_ERROR "Expected exactly one nRF clock driver")
    endif()
    set_property(TARGET drivers__clock_control PROPERTY SOURCES "${new_sources}")
    target_include_directories(drivers__clock_control PRIVATE
      "${ZEPHYR_BASE}/drivers/clock_control")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${patch_script} ${original})
  endfunction()
  cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL equinox_patch_boot_clock)
endif()

# The dual-role central only forwards peripheral events while ESB owns the radio.
if(CONFIG_ZMK_NEMU_LINK_MODE_SWITCH AND CONFIG_ZMK_SPLIT_ROLE_CENTRAL)
  set_property(GLOBAL PROPERTY EQUINOX_MODE_GUARD_SCRIPT
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch_nemu_central_forwarding.py")
  function(equinox_patch_mode_forwarding)
    get_property(mode_script GLOBAL PROPERTY EQUINOX_MODE_GUARD_SCRIPT)
    get_target_property(mode_sources app SOURCES)
    set(mode_found FALSE)
    foreach(source IN LISTS mode_sources)
      if(source MATCHES "src/split/peripheral.c$")
        set(mode_found TRUE)
        set(mode_patched "${CMAKE_CURRENT_BINARY_DIR}/equinox_split_peripheral.c")
        set(mode_original "${APPLICATION_SOURCE_DIR}/src/split/peripheral.c")
        execute_process(COMMAND ${PYTHON_EXECUTABLE} ${mode_script} ${mode_original} ${mode_patched}
                        RESULT_VARIABLE mode_result)
        if(NOT mode_result EQUAL 0)
          message(FATAL_ERROR "Equinox dual-role mode guard requires review")
        endif()
        list(REMOVE_ITEM mode_sources ${source})
        list(APPEND mode_sources ${mode_patched})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${mode_original} ${mode_script})
      endif()
    endforeach()
    if(NOT mode_found)
      message(FATAL_ERROR "Split peripheral source missing for mode guard")
    endif()
    set_property(TARGET app PROPERTY SOURCES ${mode_sources})
  endfunction()
  cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL equinox_patch_mode_forwarding)
endif()
