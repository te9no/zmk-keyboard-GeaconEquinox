if(CONFIG_ZMK_HIRES_DIAL)
  get_property(equinox_libs GLOBAL PROPERTY ZEPHYR_LIBS)
  set(pat_found FALSE)
  foreach(lib IN LISTS equinox_libs)
    get_target_property(pat_sources ${lib} SOURCES)
    get_target_property(pat_source_dir ${lib} SOURCE_DIR)
    foreach(source IN LISTS pat_sources)
      get_filename_component(pat_absolute "${source}" ABSOLUTE BASE_DIR "${pat_source_dir}")
      if(pat_absolute MATCHES "zmk-driver-hires-dial/src/hires_dial.c$")
        set(pat_found TRUE)
        set(patched_pat "${CMAKE_CURRENT_BINARY_DIR}/equinox_hires_dial.c")
        set(pat_script "${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch_pat_address.py")
        execute_process(COMMAND ${PYTHON_EXECUTABLE} ${pat_script} ${pat_absolute} ${patched_pat}
                        RESULT_VARIABLE pat_result)
        if(NOT pat_result EQUAL 0)
          message(FATAL_ERROR "PAT address patch requires source review")
        endif()
        list(REMOVE_ITEM pat_sources ${source})
        list(APPEND pat_sources ${patched_pat})
        set_property(TARGET ${lib} PROPERTY SOURCES ${pat_sources})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${pat_absolute} ${pat_script})
      endif()
    endforeach()
  endforeach()
  if(NOT pat_found)
    message(FATAL_ERROR "PAT driver not found for address patch")
  endif()
  # The pinned dial module uses ZMK behavior headers without exporting this path.
  zephyr_include_directories(${APPLICATION_SOURCE_DIR}/include)
  # Adapt the upstream dial API name to the pinned cormoran ZMK endpoint API.
  zephyr_compile_definitions(zmk_endpoints_selected=zmk_endpoint_get_selected)
endif()

if(CONFIG_PMW3610)
  get_target_property(equinox_app_sources app SOURCES)
  set(equinox_pmw_found FALSE)
  foreach(source IN LISTS equinox_app_sources)
    if(source MATCHES "zmk-driver-pmw3610-with-custom-studio-rpc/src/pmw3610.c$")
      set(equinox_pmw_found TRUE)
      set(patched_pmw "${CMAKE_CURRENT_BINARY_DIR}/equinox_pmw3610.c")
      execute_process(
        COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch_pmw_guard.py
                ${source} ${patched_pmw}
        RESULT_VARIABLE patch_result)
      if(NOT patch_result EQUAL 0)
        message(FATAL_ERROR "Equinox PMW guard could not be applied")
      endif()
      get_filename_component(pmw_src_dir ${source} DIRECTORY)
      target_include_directories(app PRIVATE ${pmw_src_dir})
      list(REMOVE_ITEM equinox_app_sources ${source})
      list(APPEND equinox_app_sources ${patched_pmw})
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
                   ${source} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/patch_pmw_guard.py)
    endif()
  endforeach()
  if(NOT equinox_pmw_found)
    message(FATAL_ERROR "Equinox PMW source not found; refusing an unguarded build")
  endif()
  set_property(TARGET app PROPERTY SOURCES ${equinox_app_sources})
endif()
